VBA - filter and highlighting

Public Sub FilterAndColor_ByDeviation_Comment()
Const HDR_ROW As Long = 4
Const HDR_COMMENT As String = "Comment"
Const HDR_DEVIATION As String = "Deviation"

Dim ws As Worksheet: Set ws = ActiveSheet

Dim firstCol As Long, lastCol As Long, lastRow As Long
firstCol = FirstNonEmptyCol(ws, HDR_ROW)
If firstCol = 0 Then MsgBox "Row " & HDR_ROW & " has no headers.", vbExclamation: Exit Sub
lastCol = ws.Cells(HDR_ROW, ws.Columns.Count).End(xlToLeft).Column
lastRow = LastUsedRow(ws)
If lastRow <= HDR_ROW Then MsgBox "No data rows under the headers.", vbExclamation: Exit Sub

Dim rngHeader As Range, rngData As Range
Set rngHeader = ws.Range(ws.Cells(HDR_ROW, firstCol), ws.Cells(HDR_ROW, lastCol))
Set rngData = ws.Range(ws.Cells(HDR_ROW + 1, firstCol), ws.Cells(lastRow, lastCol))

Dim cComment As Long, cDeviation As Long
cComment = FindHeaderCol(rngHeader, HDR_COMMENT)
cDeviation = FindHeaderCol(rngHeader, HDR_DEVIATION)
If cComment * cDeviation = 0 Then
    MsgBox "Missing header(s): " & HDR_COMMENT & ", " & HDR_DEVIATION, vbExclamation
    Exit Sub
End If

cComment = firstCol + cComment - 1
cDeviation = firstCol + cDeviation - 1

If Not ws.AutoFilterMode Then rngHeader.AutoFilter
rngHeader.AutoFilter Field:=(cComment - firstCol + 1), _
Criteria1:=Array("Increase*", "Decrease*"), _
Operator:=xlFilterValues

Dim commRange As Range
Set commRange = ws.Range(ws.Cells(HDR_ROW + 1, cComment), ws.Cells(lastRow, cComment))
commRange.FormatConditions.Delete

Dim devL As String, comL As String, fGreen As String, fRed As String
devL = ColumnLetter(cDeviation)
comL = ColumnLetter(cComment)

fGreen = "=OR(AND($" & devL & (HDR_ROW + 1) & ">0,LEFT(UPPER($" & comL & (HDR_ROW + 1) & "),8)=" & Chr(34) & "INCREASE" & Chr(34) & ")," & _
"AND($" & devL & (HDR_ROW + 1) & "<0,LEFT(UPPER($" & comL & (HDR_ROW + 1) & "),8)=" & Chr(34) & "DECREASE" & Chr(34) & "))"

fRed = "=OR(AND($" & devL & (HDR_ROW + 1) & ">0,LEFT(UPPER($" & comL & (HDR_ROW + 1) & "),8)=" & Chr(34) & "DECREASE" & Chr(34) & ")," & _
"AND($" & devL & (HDR_ROW + 1) & "<0,LEFT(UPPER($" & comL & (HDR_ROW + 1) & "),8)=" & Chr(34) & "INCREASE" & Chr(34) & "))"

With commRange.FormatConditions.Add(Type:=xlExpression, Formula1:=fGreen)
    .StopIfTrue = True
    .Interior.Color = RGB(198, 239, 206)
    .Font.Color = RGB(0, 97, 0)
End With

With commRange.FormatConditions.Add(Type:=xlExpression, Formula1:=fRed)
    .StopIfTrue = True
    .Interior.Color = RGB(255, 199, 206)
    .Font.Color = RGB(156, 0, 6)
End With

MsgBox "Done: filtered '" & HDR_COMMENT & "' and highlighted cells based on Deviation.", vbInformation
End Sub

Private Function FirstNonEmptyCol(ws As Worksheet, ByVal rowN As Long) As Long
Dim c As Long
For c = 1 To ws.Columns.Count
    If LenB(ws.Cells(rowN, c).Value2) > 0 Then FirstNonEmptyCol = c: Exit Function
Next
FirstNonEmptyCol = 0
End Function

Private Function LastUsedRow(ws As Worksheet) As Long
Dim f As Range
On Error Resume Next
Set f = ws.Cells.Find(What:="*", After:=ws.Cells(1, 1), _
LookIn:=xlFormulas, LookAt:=xlPart, _
SearchOrder:=xlByRows, SearchDirection:=xlPrevious)
On Error GoTo 0
LastUsedRow = IIf(f Is Nothing, 0, f.Row)
End Function

Private Function FindHeaderCol(hdr As Range, headerText As String) As Long
Dim cell As Range
For Each cell In hdr.Cells
    If LCase$(Trim$(cell.Value2)) = LCase$(Trim$(headerText)) Then
        FindHeaderCol = cell.Column - hdr.Column + 1
        Exit Function
    End If
Next
FindHeaderCol = 0
End Function

Private Function ColumnLetter(colNum As Long) As String
ColumnLetter = Split(Cells(1, colNum).Address(False, False), "1")(0)
End Function
